cmake_minimum_required(VERSION 3.22)
project(ACELogicBridge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/refs/tags/8.0.6.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE URL_HASH SHA256=95752f8e8b37d802b4c10182bc757de7d88f0c8899e879afa8798589be306a11)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(ACELogicBridge COMPANY_NAME "ACE-Step" BUNDLE_ID "ai.acestep.logicbridge" PLUGIN_MANUFACTURER_CODE Acst PLUGIN_CODE Acbr FORMATS AU PRODUCT_NAME "ACE-Step Bridge" COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
target_sources(ACELogicBridge PRIVATE src/Plugin.cpp)
target_compile_definitions(ACELogicBridge PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(ACELogicBridge PRIVATE juce::juce_audio_utils juce::juce_cryptography PUBLIC juce::juce_recommended_config_flags)
add_executable(capture_test tests/capture_test.cpp)
target_include_directories(capture_test PRIVATE src)
enable_testing()
add_test(NAME capture COMMAND capture_test)
juce_add_console_app(processor_probe PRODUCT_NAME "Bridge Processor Probe")
target_sources(processor_probe PRIVATE tests/processor_probe.cpp)
target_compile_definitions(processor_probe PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(processor_probe PRIVATE ACELogicBridge juce::juce_audio_utils)
add_test(NAME processor COMMAND processor_probe)

if(APPLE)
  enable_language(OBJCXX)
  add_executable(au_probe tests/au_probe.mm)
  target_link_libraries(au_probe PRIVATE "-framework Cocoa" "-framework AudioToolbox")
  add_test(NAME au_binary COMMAND au_probe "$<TARGET_FILE:ACELogicBridge_AU>")
endif()
